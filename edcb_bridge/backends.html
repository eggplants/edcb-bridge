 72    async def _request(self, method: str, path: str, **kwargs: Any) -> Any:
 73        try:
 74            res = await self.client.request(method, path, **kwargs)
 75        except httpx.HTTPError as ex:
 76            msg = f"EPGStation request failed: {method} {path}: {ex!r}"
 77            raise BackendError(msg) from ex
 78        if res.is_error:
 79            msg = f"EPGStation returned HTTP {res.status_code}: {method} {path}: {res.text[:500]}"
 80            raise BackendError(msg)
 81        if res.content and res.headers.get("content-type", "").startswith("application/json"):
 82            return res.json()
 83        return None

108    def _reserve_to_edcb(self, snapshot: EpgSnapshot, r: dict) -> dict:
109        channel_id = int(r["channelId"])
110        onid, sid = channel_id // 100000, channel_id % 100000
111        service: Service | None = snapshot.services_by_id.get((onid, sid))
112        eid = EID_NONE
113        if r.get("programId") is not None and not r.get("isTimeSpecified", False):
114            _, _, eid = split_mirakurun_program_id(int(r["programId"]))
115        start = to_jst(r["startAt"])
116        is_disabled = r.get("isSkip", False) or r.get("isOverlap", False)
117        return {
118            "title": r.get("name", ""),
119            "start_time": start,
120            "start_time_epg": start,
121            "duration_second": max(int((r["endAt"] - r["startAt"]) // 1000), 0),
122            "station_name": service.name if service else "",
123            "onid": onid,
124            # サービスが見つからない場合、地デジなら TSID = ONID とみなす
125            "tsid": service.tsid if service else (onid if GR_NETWORK_ID_MIN <= onid <= GR_NETWORK_ID_MAX else 0),
126            "sid": sid,
127            "eid": eid,
128            "comment": "EPG自動予約" if r.get("ruleId") is not None else "",
129            "reserve_id": int(r["id"]),
130            # EPGStation の isConflict はチューナー不足で録画不可を意味する
131            "overlap_mode": OVERLAP_MODE_UNAVAILABLE if r.get("isConflict", False) else 0,
132            "rec_setting": self.rec_setting_from(
133                enabled=not is_disabled,
134                directory=r.get("directory"),
135                file_name_template=r.get("recordedFormat"),
136            ),
137            "rec_file_name_list": [],
138        }

162    async def add_reserve(self, snapshot: EpgSnapshot, reserve_list: list[dict]) -> None:
163        for reserve in reserve_list:
164            body: dict[str, Any] = {
165                "allowEndLack": True,
166                "saveOption": await self._save_option(reserve.get("rec_setting", {})),
167            }
168            if reserve.get("eid", EID_NONE) != EID_NONE:
169                # 番組指定予約
170                body["programId"] = mirakurun_program_id(reserve["onid"], reserve["sid"], reserve["eid"])
171            else:
172                # プログラム予約。EPGStation では時刻指定予約になる
173                start: dt.datetime = reserve["start_time"]
174                start_ms = int(start.timestamp() * 1000)
175                body["timeSpecifiedOption"] = {
176                    "name": reserve.get("title", ""),
177                    "channelId": mirakurun_service_id(reserve["onid"], reserve["sid"]),
178                    "startAt": start_ms,
179                    "endAt": start_ms + int(reserve.get("duration_second", 0)) * 1000,
180                }
181            added = await self._request("POST", "/reserves", json=body)
182            log.info("Added EPGStation reserve: %s (%s)", added, reserve.get("title", ""))
183
184            # 無効状態で追加するよう指定された場合は、追加後にスキップ状態にする (ルール予約のみ可能)
185            if not is_enabled_rec_mode(reserve.get("rec_setting", {}).get("rec_mode", 1)):
186                log.warning("EPGStation cannot add a disabled manual reservation; it was added as enabled")

188    async def chg_reserve(self, snapshot: EpgSnapshot, reserve_list: list[dict]) -> None:
189        for reserve in reserve_list:
190            reserve_id = reserve["reserve_id"]
191            current = await self._request("GET", f"/reserves/{reserve_id}", params={"isHalfWidth": "false"})
192            want_enabled = is_enabled_rec_mode(reserve.get("rec_setting", {}).get("rec_mode", 1))
193            is_rule_reserve = current.get("ruleId") is not None
194
195            # 有効 / 無効の切り替え
196            if want_enabled:
197                if current.get("isSkip", False):
198                    await self._request("DELETE", f"/reserves/{reserve_id}/skip")
199                if current.get("isOverlap", False):
200                    await self._request("DELETE", f"/reserves/{reserve_id}/overlap")
201            elif not (current.get("isSkip", False) or current.get("isOverlap", False)):
202                if not is_rule_reserve:
203                    # EPGStation の手動予約には「無効」状態がない
204                    msg = "EPGStation cannot disable a manual reservation (delete it instead)"
205                    raise BackendError(msg)
206                # ルール予約を削除するとスキップ (無効) 状態になる
207                await self._request("DELETE", f"/reserves/{reserve_id}")
208
209            # 保存先の変更 (EPGStation で更新できるのは手動予約のみ)
210            if not is_rule_reserve:
211                save_option = await self._save_option(reserve.get("rec_setting", {}))
212                if save_option.get("directory") != current.get("directory") or save_option.get(
213                    "recordedFormat"
214                ) != current.get("recordedFormat"):
215                    await self._request(
216                        "PUT",
217                        f"/reserves/{reserve_id}",
218                        json={
219                            "allowEndLack": current.get("allowEndLack", True),
220                            "saveOption": save_option,
221                        },
222                    )

224    async def del_reserve(self, reserve_ids: list[int]) -> None:
225        for reserve_id in reserve_ids:
226            # ルールによる予約の場合、EPGStation では削除ではなくスキップ (無効) 状態になる
227            await self._request("DELETE", f"/reserves/{reserve_id}")

229    async def _get_recording(self) -> list[dict]:
230        async with self._recording_lock:
231            now = time.monotonic()
232            if self._recording_cache is not None and now - self._recording_cache[0] < RECORDING_CACHE_TTL:
233                return self._recording_cache[1]
234            data = await self._request("GET", "/recording", params={"isHalfWidth": "false", "limit": 1000})
235            self._recording_cache = (now, data["records"])
236            return data["records"]

238    async def get_recording_file_path(self, snapshot: EpgSnapshot, reserve_id: int) -> str | None:
239        records = await self._get_recording()
240        if not records:
241            return None
242        try:
243            reserve = await self._request("GET", f"/reserves/{reserve_id}", params={"isHalfWidth": "false"})
244        except BackendError:
245            return None
246        for record in records:
247            same_program = reserve.get("programId") is not None and record.get("programId") == reserve["programId"]
248            same_time = record.get("channelId") == reserve["channelId"] and record["startAt"] == reserve["startAt"]
249            if same_program or same_time:
250                video_files = record.get("videoFiles") or []
251                if video_files and video_files[0].get("filename"):
252                    return f"{self.config.recorded_dir}/{video_files[0]['filename']}"
253                return f"{self.config.recorded_dir}/{record.get('name', '')}"
254        return None

300    async def _auto_add_to_rule(self, snapshot: EpgSnapshot, data: dict, current: dict | None = None) -> dict:
301        key: dict = data["search_info"]
302        rec_setting: dict = data.get("rec_setting", {})
303
304        warn_unsupported_search_conditions(key)
305        title_only = key.get("title_only_flag", False)
306        search_option: dict[str, Any] = {
307            "keyword": key.get("and_key", ""),
308            "ignoreKeyword": strip_note(key.get("not_key", "")),
309            "keyCS": key.get("case_sensitive", False),
310            "keyRegExp": key.get("reg_exp_flag", False),
311            "name": True,
312            "description": not title_only,
313            "extended": not title_only,
314            "ignoreKeyCS": key.get("case_sensitive", False),
315            "ignoreKeyRegExp": key.get("reg_exp_flag", False),
316            "ignoreName": True,
317            "ignoreDescription": not title_only,
318            "ignoreExtended": not title_only,
319            **service_list_to_search_option(snapshot, key.get("service_list", [])),
320        }
321        if genres := content_list_to_genres(key.get("content_list", [])):
322            search_option["genres"] = genres
323        if times := date_list_to_times(key.get("date_list", [])):
324            search_option["times"] = times
325        if key.get("free_ca_flag") == FREE_CA_FREE_ONLY:
326            search_option["isFree"] = True
327        if key.get("chk_duration_min", 0) > 0:
328            search_option["durationMin"] = key["chk_duration_min"] * 60
329        if key.get("chk_duration_max", 0) > 0:
330            search_option["durationMax"] = key["chk_duration_max"] * 60
331
332        reserve_option = build_reserve_option(key, rec_setting)
333
334        rule: dict[str, Any] = {
335            "isTimeSpecification": False,
336            "searchOption": search_option,
337            "reserveOption": reserve_option,
338            "saveOption": await self._save_option(rec_setting),
339        }
340        # EDCB 側にない項目 (タグ・エンコード設定) は既存ルールの値を引き継ぐ
341        if current is not None:
342            if "tags" in (current.get("reserveOption") or {}):
343                reserve_option["tags"] = current["reserveOption"]["tags"]
344            if current.get("encodeOption"):
345                rule["encodeOption"] = current["encodeOption"]
346        return rule

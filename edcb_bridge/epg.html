 61def guess_tsid(channel_type: str, channel: str, onid: int) -> int:
 62    """
 63    mirakc の API は TSID を返さないため、チャンネル種別とチャンネル名から推測する
 64
 65    - 地デジ: TSID は原則ネットワーク ID と同じ
 66    - BS: "BS15_1" のような (トランスポンダ番号, スロット番号) 形式から算出 (0x4000 | tp << 4 | slot)
 67    - CS: "CS4" / "ND04" のようなトランスポンダ番号から算出 (0x6000 | nd << 4)
 68    """
 69    if channel_type == "GR":
 70        return onid
 71    if channel_type == "BS":
 72        m = re.fullmatch(r"BS(\d+)_(\d+)", channel)
 73        if m:
 74            return 0x4000 | int(m.group(1)) << 4 | int(m.group(2))
 75    if channel_type == "CS":
 76        m = re.fullmatch(r"(?:CS|ND)(\d+)", channel)
 77        if m:
 78            return 0x6000 | int(m.group(1)) << 4
 79    # 推測できない場合は 0 (KonomiTV は TSID なしのチャンネルでは予約できない)
 80    log.warning("Cannot determine TSID: type=%s channel=%s onid=%d", channel_type, channel, onid)
 81    return 0

 88def filetime_to_datetime(ft: int) -> dt.datetime:
 89    """
 90    EDCB の FILETIME 値を datetime に変換する
 91    EDCB の FILETIME は「JST のローカル時刻を UTC とみなした」値なので、9時間を引いて実時刻に戻す
 92    """
 93    unix_sec = (ft - 116444736000000000) / 10000000 - 9 * 3600
 94    try:
 95        return dt.datetime.fromtimestamp(unix_sec, tz=JST)
 96    except (OverflowError, OSError, ValueError):
 97        # 0x7fffffffffffffff (無限大) や 1 (無限小) のような範囲外の値
 98        return dt.datetime.max.replace(tzinfo=JST) if unix_sec > 0 else dt.datetime.min.replace(tzinfo=JST)

147class EpgSource:
148    """mirakc からサービス・番組情報を取得してキャッシュする"""
149
150    def __init__(self, config: Config, client: httpx.AsyncClient) -> None:
151        self.config = config
152        self.client = client
153        self.snapshot = EpgSnapshot()
154        self.lock = asyncio.Lock()
155
156    async def get(self) -> EpgSnapshot:
157        async with self.lock:
158            fetched_at = self.snapshot.fetched_at
159            if fetched_at is not None and time.monotonic() - fetched_at < self.config.epg_cache_ttl:
160                return self.snapshot
161            try:
162                self.snapshot = await self._fetch()
163            except Exception:
164                # 取得に失敗しても、古いキャッシュがあればそれを返す
165                if fetched_at is None:
166                    raise
167                log.exception("Failed to refresh EPG from mirakc, using stale cache")
168            return self.snapshot
169
170    async def _fetch(self) -> EpgSnapshot:
171        base = self.config.mirakc_url
172        channels_res, services_res, programs_res = await asyncio.gather(
173            self.client.get(f"{base}/api/channels"),
174            self.client.get(f"{base}/api/services"),
175            self.client.get(f"{base}/api/programs"),
176        )
177        for res in (channels_res, services_res, programs_res):
178            res.raise_for_status()
179
180        # チャンネル (物理チャンネル) 名を引けるようにしておく
181        channel_names: dict[tuple[str, str], str] = {}
182        for ch in channels_res.json():
183            channel_names[(ch["type"], ch["channel"])] = ch.get("name", "")
184
185        snapshot = EpgSnapshot(fetched_at=time.monotonic())
186        for s in services_res.json():
187            ch = s.get("channel") or {}
188            ch_type = ch.get("type", "")
189            ch_name = ch.get("channel", "")
190            onid = int(s["networkId"])
191            tsid = (
192                int(s["transportStreamId"])
193                if s.get("transportStreamId") is not None
194                else guess_tsid(ch_type, ch_name, onid)
195            )
196            service = Service(
197                onid=onid,
198                tsid=tsid,
199                sid=int(s["serviceId"]),
200                service_type=int(s.get("type", 1)),
201                name=s.get("name", ""),
202                channel_type=ch_type,
203                channel=ch_name,
204                network_name=channel_names.get((ch_type, ch_name), ch_type),
205                remote_control_key_id=int(s.get("remoteControlKeyId") or 0),
206            )
207            snapshot.services.append(service)
208            snapshot.services_by_id[(service.onid, service.sid)] = service
209
210        for p in programs_res.json():
211            snapshot.programs.setdefault((int(p["networkId"]), int(p["serviceId"])), []).append(p)
212        for plist in snapshot.programs.values():
213            plist.sort(key=lambda p: p.get("startAt", 0))
214
215        log.info(
216            "EPG refreshed: %d services, %d programs",
217            len(snapshot.services),
218            sum(len(v) for v in snapshot.programs.values()),
219        )
220        return snapshot

223def build_chset5(snapshot: EpgSnapshot) -> bytes:
224    """EDCB の ChSet5.txt を生成する (UTF-8 BOM 付き、CRLF)"""
225    lines = []
226    for s in snapshot.services:
227        # 列: サービス名, ネットワーク名, ONID, TSID, SID, サービスタイプ,
228        #     部分受信, EPG 取得, 検索対象, リモコン ID
229        is_partial = s.service_type == ONESEG_SERVICE_TYPE
230        lines.append(
231            "\t".join(
232                [
233                    s.name,
234                    s.network_name,
235                    str(s.onid),
236                    str(s.tsid),
237                    str(s.sid),
238                    str(s.service_type),
239                    "1" if is_partial else "0",
240                    "1",
241                    "0" if is_partial else "1",
242                    str(s.remote_control_key_id),
243                ]
244            )
245        )
246    return "﻿".encode() + ("\r\n".join(lines) + "\r\n").encode("utf-8")

256def program_to_event_info(service: Service, p: dict[str, Any]) -> dict:
257    """mirakc の番組情報を EDCB の EventInfo に変換する"""
258    v: dict[str, Any] = {
259        "onid": service.onid,
260        "tsid": service.tsid,
261        "sid": service.sid,
262        "eid": int(p["eventId"]),
263        "free_ca_flag": 0 if p.get("isFree", True) else 1,
264    }
265    if p.get("startAt") is not None:
266        v["start_time"] = to_jst(p["startAt"])
267    # 放送時間未定の番組では duration が 0 や巨大な値になることがある
268    duration = p.get("duration")
269    if duration is not None and 0 < duration < 86400 * 1000 * 7:
270        v["duration_sec"] = int(duration // 1000)
271
272    if p.get("name") is not None:
273        v["short_info"] = {"event_name": p.get("name", ""), "text_char": p.get("description", "") or ""}
274    ext = build_extended_text(p.get("extended"))
275    if ext:
276        v["ext_info"] = {"text_char": ext}
277
278    genres = p.get("genres") or []
279    if genres:
280        v["content_info"] = {
281            "nibble_list": [
282                {
283                    "content_nibble": (g.get("lv1", 0xF) & 0xF) << 8 | (g.get("lv2", 0xF) & 0xF),
284                    "user_nibble": (g.get("un1", 0xF) & 0xF) << 8 | (g.get("un2", 0xF) & 0xF),
285                }
286                for g in genres
287            ]
288        }
289
290    video = p.get("video")
291    if video and video.get("streamContent") is not None:
292        v["component_info"] = {
293            "stream_content": video.get("streamContent", 1),
294            "component_type": video.get("componentType", 0),
295            "component_tag": 0,
296            "text_char": "",
297        }
298
299    audios = p.get("audios") or ([p["audio"]] if p.get("audio") else [])
300    if audios:
301        v["audio_info"] = {
302            "component_list": [
303                {
304                    "stream_content": 2,
305                    "component_type": a.get("componentType", 3),
306                    "component_tag": 0x10 + i,
307                    "stream_type": 0x0F,
308                    "simulcast_group_tag": 0xFF,
309                    "es_multi_lingual_flag": 1 if len(a.get("langs") or []) > 1 else 0,
310                    "main_component_flag": 1 if a.get("isMain", i == 0) else 0,
311                    "quality_indicator": 3,
312                    "sampling_rate": SAMPLING_RATE_CODES.get(a.get("samplingRate", 48000), 7),
313                    "text_char": "",
314                }
315                for i, a in enumerate(audios)
316            ]
317        }
318
319    # イベント共有 (サブチャンネルでの同一番組) とイベントリレー
320    ## KonomiTV は「要素が 1 つだけで、それが自分自身でない」グループを持つ番組をスキップするため、自分自身も含めて返す
321    shared = [x for x in (p.get("relatedItems") or []) if x.get("type") == "shared"]
322    if shared:
323        v["event_group_info"] = {
324            "group_type": 1,
325            "event_data_list": [
326                {
327                    "onid": x.get("networkId") or service.onid,
328                    "tsid": service.tsid,
329                    "sid": x["serviceId"],
330                    "eid": x["eventId"],
331                }
332                for x in shared
333            ],
334        }
335    relay = [x for x in (p.get("relatedItems") or []) if x.get("type") == "relay"]
336    if relay:
337        v["event_relay_info"] = {
338            "group_type": 2,
339            "event_data_list": [
340                {
341                    "onid": x.get("networkId") or service.onid,
342                    "tsid": service.tsid,
343                    "sid": x["serviceId"],
344                    "eid": x["eventId"],
345                }
346                for x in relay
347            ],
348        }
349    return v

352def enum_pg_info(snapshot: EpgSnapshot, key_list: list[int]) -> list[dict]:
353    """
354    CMD_EPG_SRV_ENUM_PG_INFO_EX の実装
355    key_list の最終 2 要素は開始時刻の範囲 (FILETIME)、それ以外は (OR マスク, サービスキー) のペア
356    """
357    # 最低でも時間範囲の 2 要素が必要で、全体は常にペアの並びになる
358    if len(key_list) < MIN_PG_INFO_KEY_LIST_LENGTH or len(key_list) % 2 != 0:
359        return []
360    start_min = filetime_to_datetime(key_list[-2])
361    start_max = filetime_to_datetime(key_list[-1])
362    pairs = [(key_list[i], key_list[i + 1]) for i in range(0, len(key_list) - 2, 2)]
363
364    result = []
365    for s in snapshot.services:
366        if not any((s.key | mask) == (key | mask) for mask, key in pairs):
367            continue
368        events = []
369        for p in snapshot.programs.get((s.onid, s.sid), []):
370            if p.get("startAt") is None:
371                continue
372            start = to_jst(p["startAt"])
373            if start_min <= start < start_max:
374                events.append(program_to_event_info(s, p))
375        if events:
376            result.append({"service_info": s.to_service_info(), "event_list": events})
377    return result

151    def __init__(self, config: Config, client: httpx.AsyncClient) -> None:
152        self.config = config
153        self.client = client
154        self.snapshot = EpgSnapshot()
155        self.lock = asyncio.Lock()
156
157    async def get(self) -> EpgSnapshot:
158        async with self.lock:
159            fetched_at = self.snapshot.fetched_at
160            if fetched_at is not None and time.monotonic() - fetched_at < self.config.epg_cache_ttl:
161                return self.snapshot
162            try:
163                self.snapshot = await self._fetch()
164            except Exception:
165                # 取得に失敗しても、古いキャッシュがあればそれを返す
166                if fetched_at is None:
167                    raise
168                log.exception("Failed to refresh EPG from mirakc, using stale cache")
169            return self.snapshot
170
171    async def _fetch(self) -> EpgSnapshot:
172        base = self.config.mirakc_url
173        channels_res, services_res, programs_res = await asyncio.gather(
174            self.client.get(f"{base}/api/channels"),
175            self.client.get(f"{base}/api/services"),
176            self.client.get(f"{base}/api/programs"),
177        )
178        for res in (channels_res, services_res, programs_res):
179            res.raise_for_status()
180
181        # チャンネル (物理チャンネル) 名を引けるようにしておく
182        channel_names: dict[tuple[str, str], str] = {}
183        for ch in channels_res.json():
184            channel_names[(ch["type"], ch["channel"])] = ch.get("name", "")
185
186        snapshot = EpgSnapshot(fetched_at=time.monotonic())
187        for s in services_res.json():
188            ch = s.get("channel") or {}
189            ch_type = ch.get("type", "")
190            ch_name = ch.get("channel", "")
191            onid = int(s["networkId"])
192            tsid = (
193                int(s["transportStreamId"])
194                if s.get("transportStreamId") is not None
195                else guess_tsid(ch_type, ch_name, onid)
196            )
197            service = Service(
198                onid=onid,
199                tsid=tsid,
200                sid=int(s["serviceId"]),
201                service_type=int(s.get("type", 1)),
202                name=s.get("name", ""),
203                channel_type=ch_type,
204                channel=ch_name,
205                network_name=channel_names.get((ch_type, ch_name), ch_type),
206                remote_control_key_id=int(s.get("remoteControlKeyId") or 0),
207            )
208            snapshot.services.append(service)
209            snapshot.services_by_id[(service.onid, service.sid)] = service
210
211        for p in programs_res.json():
212            snapshot.programs.setdefault((int(p["networkId"]), int(p["serviceId"])), []).append(p)
213        for plist in snapshot.programs.values():
214            plist.sort(key=lambda p: p.get("startAt", 0))
215
216        log.info(
217            "EPG refreshed: %d services, %d programs",
218            len(snapshot.services),
219            sum(len(v) for v in snapshot.programs.values()),
220        )
221        return snapshot

def enum_pg_info(snapshot: EpgSnapshot, key_list: list[int]) -> list[dict]: 
353def enum_pg_info(snapshot: EpgSnapshot, key_list: list[int]) -> list[dict]:
354    """
355    CMD_EPG_SRV_ENUM_PG_INFO_EX の実装
356    key_list の最終 2 要素は開始時刻の範囲 (FILETIME)、それ以外は (OR マスク, サービスキー) のペア
357    """
358    # 最低でも時間範囲の 2 要素が必要で、全体は常にペアの並びになる
359    if len(key_list) < MIN_PG_INFO_KEY_LIST_LENGTH or len(key_list) % 2 != 0:
360        return []
361    start_min = filetime_to_datetime(key_list[-2])
362    start_max = filetime_to_datetime(key_list[-1])
363    pairs = [(key_list[i], key_list[i + 1]) for i in range(0, len(key_list) - 2, 2)]
364
365    result = []
366    for s in snapshot.services:
367        if not any((s.key | mask) == (key | mask) for mask, key in pairs):
368            continue
369        events = []
370        for p in snapshot.programs.get((s.onid, s.sid), []):
371            if p.get("startAt") is None:
372                continue
373            start = to_jst(p["startAt"])
374            if start_min <= start < start_max:
375                events.append(program_to_event_info(s, p))
376        if events:
377            result.append({"service_info": s.to_service_info(), "event_list": events})
378    return result
